function weeklyAttCalendarRows71_() {
  /* 行事曆雲端格式：pull('calendar') → meta.totalChunks；每塊 records[0].db = { '2026': [假日列…] } */
  var out = [];
  try {
    var p = pullPayload('calendar') || {}, m = p.meta || {};
    var n = Number(m.totalChunks) || (m.smartBuckets || p.smartBuckets || []).length || Number(p.chunks) || 1;
    for (var i = 0; i < n && i < 80; i++) {
      var c = pullPayload('calendar', i);
      ((c && c.records) || []).forEach(function (rec) {
        if (rec && rec.db && typeof rec.db === 'object') Object.keys(rec.db).forEach(function (y) { if (Array.isArray(rec.db[y])) out = out.concat(rec.db[y]); });
        else if (rec && rec.date) out.push(rec);
      });
    }
  } catch (e) { console.warn('[ATT-WEEKLY] calendar load failed: ' + e); }
  return out;
}
function weeklyAttendanceRun71_(preview) {
  var tz = HRA_CFG.TZ, today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  var t = new Date(today + 'T00:00:00Z'), dow = t.getUTCDay();
  var mon = new Date(t); mon.setUTCDate(t.getUTCDate() - ((dow + 6) % 7) - 7);
  var key = mon.toISOString().slice(0, 10), guard = 'att_weekly_sent_' + key, dates = [];
  for (var i = 0; i < 7; i++) { var d = new Date(mon); d.setUTCDate(mon.getUTCDate() + i); dates.push(d.toISOString().slice(0, 10)); }
  var off = new Set(); try { off = HRACalendarModel.holidayDates(weeklyAttCalendarRows71_()); } catch (e) { console.warn('[ATT-WEEKLY] holiday calc failed: ' + e); }
  var workday = dow !== 0 && !off.has(today);
  var by = {}, seen = {};
  attendanceRecordsForDates(dates).forEach(function (r) {
    var dt = attendanceDateKey(r); if (!dt) return;
    var dims = [r.dept, r.sect, r.line].map(function (x) { return String(x == null ? '' : x).trim().toUpperCase().replace(/\s+/g, ' '); }).join('|');
    var kind = r.isGrand ? 'G' : r.isSubtotal ? 'S' : 'D', k = dt + '|' + kind + '|' + dims;
    by[dt] = by[dt] || [];
    if (dims.replace(/\|/g, '') && seen[k] !== undefined) { by[dt][seen[k]] = r; return; }
    seen[k] = by[dt].length; by[dt].push(r);
  });
  var days = Object.keys(by).sort().map(function (dt) { return { date: dt, rows: by[dt] }; }).filter(function (x) { return x.rows.some(function (r) { return r.isGrand; }); });
  var label = days.length ? days[0].date.slice(5).replace('-', '/') + ' ~ ' + days[days.length - 1].date.slice(5).replace('-', '/') : key;
  var text = HRAAttTg.build({ mode: 'weekly', lang: 'both', key: key, periodLabel: label, days: days });
  if (preview) return { today: today, week: key, workday: workday, holidays: off.size, next: Array.from(off).filter(function (x) { return x >= today; }).sort().slice(0, 6), days: days.length, alreadySent: !!cfg(guard), text: text };
  if (cfg(guard)) return { skipped: 'sent' };
  if (!workday) { console.log('[ATT-WEEKLY] ' + today + ' is not a working day; wait'); return { skipped: 'holiday' }; }
  if (!days.length) { console.log('[ATT-WEEKLY] no attendance data for week ' + key); return { skipped: 'nodata' }; }
  var sent = tgSend(HRA_CFG.CHAT_ID, text, summaryKeyboard('zh', 'attendance'));
  if (sent && sent.message_id) P().setProperty(guard, nowIso());
  return { sent: !!(sent && sent.message_id), week: key };
}
function ensureWeeklyAttTrigger71_() {
  if (cfg('att_weekly_trigger71')) return;
  try {
    ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'weeklyAttendanceJob') ScriptApp.deleteTrigger(t); });
    ScriptApp.newTrigger('weeklyAttendanceJob').timeBased().everyDays(1).atHour(8).inTimezone(HRA_CFG.TZ).create();
    P().setProperty('att_weekly_trigger71', nowIso());
  } catch (e) { console.warn('[ATT-WEEKLY] trigger install failed: ' + e); }
}
